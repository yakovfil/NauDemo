cmake_minimum_required(VERSION 3.24)
get_filename_component(demo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(engine "${demo}/../NauEngineLite" ABSOLUTE)
get_filename_component(modules "${demo}/../NauEngineModules" ABSOLUTE)
project(NauDemoScenePortability LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(NAU_RUNTIME_PROFILE scene CACHE STRING "CPU scene validation" FORCE)
set(NAU_EXTERNAL_RUNTIME_FILE "${demo}/cmake/NauDemoRuntime.cmake")
set(NAU_MODULES_SOURCE_DIR "${modules}")
list(APPEND NAU_EXTERNAL_PROJECT_DIRS "${modules}")
set(NAU_CORE_SAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${engine}" engine)
set(gtest_force_shared_crt OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
add_subdirectory("${engine}/engine/3rdparty_libs/googletest" googletest)
list(APPEND CMAKE_MODULE_PATH "${engine}/cmake")
include(NauCommon)
include(NauModule)
include(platformSpec/NauPlatformSetup)
nau_target_add_compile_options(TARGETS gtest gtest_main)
add_executable(NauDemoSceneTests "${demo}/tests/test_demo_portability.cpp")
target_include_directories(NauDemoSceneTests PRIVATE
  "${demo}/src")
target_include_directories(NauDemoSceneTests PRIVATE
  "${modules}/modules/scene/src"
  "${modules}/modules/assets/src")
target_link_libraries(NauDemoSceneTests PRIVATE CoreScene CoreAssets gtest_main)
target_link_libraries(NauDemoSceneTests PRIVATE CoreAssetsCpu)
target_link_libraries(NauDemoSceneTests PRIVATE NauScreenText)
target_sources(NauDemoSceneTests PRIVATE "${modules}/modules/screen_text/tests/test_screen_text.cpp")
target_link_libraries(NauDemoSceneTests PRIVATE NauFrameworkScene)
nau_target_link_modules(NauDemoSceneTests CoreScene CoreAssets CoreAssetsCpu)
target_sources(NauDemoSceneTests PRIVATE "${demo}/tests/test_demo_scene_lifecycle.cpp")
target_sources(NauDemoSceneTests PRIVATE "${modules}/modules/assets_cpu/tests/test_cpu_mesh.cpp")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
foreach(input IN ITEMS cube.gltf cube.bin)
  configure_file("${demo}/assets/${input}" "${CMAKE_BINARY_DIR}/fixtures/${input}" COPYONLY)
endforeach()
configure_file("${demo}/assets/fonts/Abel-Regular.ttf" "${CMAKE_BINARY_DIR}/fixtures/Abel-Regular.ttf" COPYONLY)
if(EMSCRIPTEN)
  target_compile_definitions(NauDemoSceneTests PRIVATE NAU_TEXT_FIXTURE_DIR="/fixtures")
else()
  target_compile_definitions(NauDemoSceneTests PRIVATE NAU_TEXT_FIXTURE_DIR="${CMAKE_BINARY_DIR}/fixtures")
endif()
nau_add_compile_options(NauDemoSceneTests)
include("${CMAKE_CURRENT_SOURCE_DIR}/mesh_sources.cmake")
enable_testing()
if(EMSCRIPTEN)
  target_link_options(NauDemoSceneTests PRIVATE -sPTHREAD_POOL_SIZE=12 -sINITIAL_MEMORY=67108864)
  target_link_options(NauDemoSceneTests PRIVATE "SHELL:--preload-file ${CMAKE_BINARY_DIR}/fixtures@/fixtures")
  target_link_options(NauDemoSceneTests PRIVATE -sEXIT_RUNTIME=1)
  add_test(NAME DemoSceneCpu COMMAND "$ENV{EMSDK_NODE}" $<TARGET_FILE:NauDemoSceneTests>)
else()
  add_test(NAME DemoSceneCpu COMMAND NauDemoSceneTests)
endif()
set_tests_properties(DemoSceneCpu PROPERTIES TIMEOUT 30)
